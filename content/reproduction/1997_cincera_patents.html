Get the data

The data set can be downloaded from the Journal of Applied Econometrics Data Archive.

download.file("http://qed.econ.queensu.ca/jae/1997-v12.3/cincera/mc-data.zip",
              destfile = "1997_cincera_data.zip")
unzip("1997_cincera_data.zip")

raw_data <- read.delim("data.mc", header = FALSE)
names(raw_data) <- c("fi", "s", "g",
                     "p83", "p84", "p85", "p86", "p87",
                     "p88", "p89", "p90", "p91",
                     "lr83", "lr84", "lr85", "lr86", "lr87",
                     "lr88", "lr89", "lr90", "lr91",
                     "ls83", "ls84", "ls85", "ls86", "ls87",
                     "ls88", "ls89", "ls90", "ls91")

years <- 83:91

p <- matrix(as.matrix(raw_data[, paste0("p", years)]))
k <- matrix(as.matrix(raw_data[, paste0("lr", years)]))
spill <- matrix(as.matrix(raw_data[, paste0("ls", years)]))

# Firm IDs
fi <- rep(1:181, 9)

# Years
year <- c()
for (i in 1983:1991) {
  year <- append(year, rep(i, 181))
}

# Generate the final data frame
patents <- data.frame(year, fi, p, k)

# Lags of R&D Spending
for (i in 1:4) {
  n_lag <- i * 181
  patents[[paste0("k.", i)]] <- c(rep(NA, n_lag), k[1:(length(k) - n_lag)])
}

# Spillovers and their lags
patents[["spill"]] <- as.vector(spill)

for (i in 1:4) {
  n_lag <- i * 181
  patents[[paste0("spill.", i)]] <- c(rep(NA, n_lag),
                                      spill[1:(length(spill) - n_lag)])
}

# Geographic dummies
g <- rep(raw_data[["g"]], 9)

for (i in 1:4) {
  patents[[paste0("g.", i)]] <- as.numeric(g == i)
}

# Technological dummies
s <- rep(raw_data[["s"]], 9)

for (i in 1:15) {
  patents[[paste0("s.", i)]] <- as.numeric(s == i)
}

# Give labels to the variables
attr(patents, "var.labels") <- c("Year", "FirmID", "# of patents",
                                 "R&D spending", "Lag R&D 1", "Lag R&D 2",
                                 "Lag R&D 3", "Lag R&D 4", "Spillover",
                                 "Lag spillover 1", "Lag spillover 2",
                                 "Lag spillover 3", "Lag spillover 4",
                                 "Geographic dummy 1", "Geographic dummy 2",
                                 "Geographic dummy 3", "Geographic dummy 4",
                                 "Sector dummy 1", "Sector dummy 2",
                                 "Sector dummy 3", "Sector dummy 4",
                                 "Sector dummy 5", "Sector dummy 6",
                                 "Sector dummy 7", "Sector dummy 8",
                                 "Sector dummy 9", "Sector dummy 10",
                                 "Sector dummy 11", "Sector dummy 12",

Panel estimates

Poisson

Cincera (1997) estimates a simple Poisson model for panel data as a benchmark. In R this can be done with the pglm package for general linear models for panel data.

library("pglm")

# Table 2
# (3) Conditional Poisson
model_poisson <- pglm(p ~ k + k.1 + k.2 + k.3 + k.4 +
                        spill + spill.1 + spill.2 + spill.3 + spill.4,
                      data = patents,
                      model = "within",
                      effect = "individual",
                      family = poisson())

summary(model_poisson)
