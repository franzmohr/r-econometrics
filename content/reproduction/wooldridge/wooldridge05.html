data("bwght")

bwght_half <- bwght[1:694, ]

model_1 <- lm(lbwght ~ cigs + lfaminc, data = bwght_half)
summary_1 <- summary(model_1)

model_2 <- lm(lbwght ~ cigs + lfaminc, data = bwght)
summary_2 <- summary(model_2)

summary_2[["coefficients"]][2, 2] / summary_1[["coefficients"]][2, 2]
## [1] 0.6443341
model_3 <- lm(cigs ~ lfaminc, data = bwght_half)
summary_3 <- summary(model_3)

sigma_j <- summary_3[["coefficients"]][2, 2]
sigma_hat <- summary_3[["sigma"]]
r2 <- summary_3[["r.squared"]]
sigma_hat / (sqrt(1388) * sigma_j * sqrt(1 - r2))
## [1] 0.6609623
